export type ResultStatus = 'SUCCESS' | 'ERROR';

export interface AnalysisResultPayload {
    status: ResultStatus;
    message?: string;
    analysisId?: string;
}

export class AnalysisResult {
    public readonly status: ResultStatus;
    public readonly message: string | null;
    public readonly analysisId: string | null;

    constructor(payload: AnalysisResultPayload) {
        this.status = payload.status;
        this.message = payload.message;
        this.analysisId = payload.analysisId;
    }

    public isSuccess(): boolean {
        return this.status === 'SUCCESS';
    }
}

export class StartAnalysisService implements StartAnalysisUseCase {
    
    public constructor(
        private readonly provider: AnalysisProvider,
        // Iniezione delle Porte
        private readonly credentialPort: IGitCredentialPort,
        private readonly githubAccessPort: IGitHubAccessPort,
        private readonly gitClonePort: IGitClonePort,
        private readonly storagePort: IStoragePort,
        private readonly orchestratorPort: IOrchestratorPort
    ) {}

    private createAnalysis(command: StartAnalysisCommand): GitHubAnalysis {
        return this.provider.createGitHubAnalysisEntity(command);
    }

    async execute(command: StartAnalysisCommand): Promise<AnalysisResult> {
        // 1. Creo la Entity
        const analysis = this.createAnalysis(command);
        let actualPatToken: string | null = null;

        // 2. Controllo il PAT (se fornito)
        if (command.patPassword) {
            const patRequest = new ValidatePatRequestModel(command.repositoryUrl, command.patPassword);
            const patResponse = await this.credentialPort.validate(patRequest);

            if (!patResponse.isValid) {
                return new AnalysisResult({ status: 'ERROR', message: 'PAT_INVALID_OR_NOT_FOUND' });
            }
            else actualPatToken = patResponse.token;
        }

        // 3. Chiedo a GitHub se la repo è accessibile
        const accessRequest = new CheckAvailabilityRequestModel(
            command.repositoryUrl,
            actualPatToken,
            command.branch,
            command.commitHash || null
        );
        const accessResponse = await this.githubAccessPort.check(accessRequest);

        if (!accessResponse.isAccessible) {
            return new AnalysisResult({ status: 'ERROR', message: accessResponse.errorMessage || 'REPO_NOT_ACCESSIBLE' });
        }

        // 4. Clono la repository
        // [SUGGERIMENTO SERVICE]: Se la logica di gestione della cartella locale, 
        // pulizia dei file temporanei o gestione degli errori git diventa complessa, 
        // estrarrei questo blocco in un `Service`.
        const cloneRequest = new CloneRepoRequestModel(
            command.repositoryUrl,
            analysis.getAnalysisId().value, // Usiamo l'ID dell'analisi per nominare la cartella isolata
            actualPatToken,
            command.branch,
            command.commitHash || null
        );
        const cloneResponse = await this.gitClonePort.clone(cloneRequest);

        if (!cloneResponse.success || !cloneResponse.localFolderPath) {
            return new AnalysisResult({ status: 'ERROR', message: 'CLONE_FAILED' });
        }

        // 5. Salvo su S3
        // [SUGGERIMENTO SERVICE]: Se prima di caricare su S3 devi zippare la cartella, 
        // rimuovere la cartella .git o offuscare dati, crea un `ArchiveAndUploadService`.
        const s3Key = `analyses/${command.userId}/${analysis.getId()}`;
        const uploadRequest = new UploadToStorageRequestModel(cloneResponse.localFolderPath, s3Key);
        const uploadResponse = await this.storagePort.upload(uploadRequest);

        if (!uploadResponse.success) {
            // Qui potresti voler pulire la cartella locale (rollback parziale)
            return new AnalysisResult({ status: 'ERROR', message: 'UPLOAD_TO_S3_FAILED' });
        }

        // 6. Procediamo: Aggiorno l'entity con l'URL di S3 e la salvo sul DB
        analysis.setS3SourceUrl(uploadResponse.s3Url);
        await this.provider.saveAnalysis(analysis);

        const analyzeRequest = new AnalyzeRequestModel();
        const startAnalyzeResponse = await this.orchestratorPort.analyze(analyzeRequest);

        // Ritorno il risultato di successo
        return new AnalysisResult({ status: 'SUCCESS', analysisId: analysis.getId() });
    }
}
